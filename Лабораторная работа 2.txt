int z_1()
{
	int a = 1;
	int b = 2;
	int c = 3;
	int d = 4;
	printf(" %d\n \t%d\n \t\t%d\n \t\t\t%d\n", a, b, c, d);
	printf(" %-10.5f\n", 12.234657);
	printf(" Остаток от деления %d на %d равен %d\n ", 5, 2, 5 % 2);
	printf("%1.0f разделить на %1.0f равно %1.2f\n ", 7., 5., 7. / 5.);
	printf("%d умножить на %d равно %d\n ", 2000, 4, 2000 * 4);
	printf("%g разделить на %e равно %f\n\n ", 5.,2000000., 5. / 2000000);
	return 0;
	getchar(); 
}
int z_2()
{
	float N = 15;
	float K = 10;
	printf("Сейчас %1.0f часов %1.0f минут 00 секунд\n", N, K);
	printf(" Идет %1.0f минута суток\n", N * 60 + K);
	printf(" До полуночи осталось %1.0f часов и %1.0f минут\n", 24. - N, 60. - K);
	printf(" С 8.00 прошло %1.0f секунд\n", (N - 8.) * 3600 + K * 60);
	printf(" Текущий час  = %1.2f суток  и текущая минута = %1.2f часа\n\n", N / 24., K / 60.);
	return 0;
	getchar(); 
}
