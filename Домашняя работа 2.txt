#define _CRT_SECURE_NO_DEPRECATE
#define _USE_MATH_DEFINES 
#include <locale.h>
#include <stdio.h>
#include <math.h>
#define M_PI 3.14159265358979323846
float dz(float h1, float h2, float d, float L);

//h1 - расстояние от реки до костра (h1 ≥ 0). 
//h2 – расстояние от реки до человека (h2 ≥ 0). 
//d – расстояние от костра до человека вдоль течения реки (d ≥ 0). 
//L – кратчайший путь
//Определить минимальное расстояние, которое должен пройти человек, до костра, зачерпнув по дороге к нему воды в реке: L = (d**2 + (h1+h2)**2)**1/2

float main()
{
	setlocale(LC_ALL, "RUS");
	float result, h1, h2, d, L;
	puts("Введите расстояние от реки до костра (h1 ≥ 0)");
	scanf("%f", &h1);
	puts("Введите расстояние от реки до человека (h2 ≥ 0)");
	scanf("%f", &h2);
	puts("Введите расстояние от костра до человека вдоль течения реки (d ≥ 0)");
	scanf("%f", &d);
	printf("h1 = %.0f, h2 = %.0f, d = %.0f\n", h1, h2, d);
	result = dz(h1, h2, d, L);
	printf("Минимальное расстояние, которое должен пройти человек, до костра, зачерпнув по дороге к нему воды в реке равно %.2f метров\n", result);
	return 0;
} 

float dz(float h1, float h2, float d, float L)
{
	L = pow((pow(d, 2) + pow(h1 + h2, 2)), 1./2);
	return L;
}
